### [1] 截圖捕捉
使用 Python 的 Pillow 或 mss 套件捕捉整個螢幕畫面，輸出為 PIL Image 物件。輸入：無。輸出：PIL Image 物件，包含螢幕當前畫面。

```python
import mss
from PIL import Image

def capture_screen() -> Image.Image:
    with mss.mss() as sct:
        monitor = sct.monitors[0]  # full screen
        sct_img = sct.grab(monitor)
        img = Image.frombytes("RGB", sct_img.size, sct_img.rgb)
    return img

# Example usage:
if __name__ == "__main__":
    screenshot = capture_screen()
    screenshot.show()  # Opens the captured image in the default image viewer
    # To save the screenshot, uncomment the following line:
    # screenshot.save("screenshot.png")
```

### [2] 影像編碼
將 PIL Image 物件轉換為 JPEG 或 PNG 格式的 bytes，並附上影像尺寸與編碼參數。輸入：PIL Image。輸出：bytes、影像寬高。

```python
import io
from typing import Tuple
from PIL import Image

def encode_image(image: Image.Image, fmt: str = "JPEG", **params) -> Tuple[bytes, int, int]:
    """
    Encode a PIL Image to bytes in the specified format.

    :param image: PIL Image to encode.
    :param fmt: Target format ("JPEG", "PNG", etc.).
    :param params: Optional format-specific parameters (e.g., quality for JPEG).
    :return: Tuple of (encoded bytes, width, height).
    """
    buf = io.BytesIO()
    image.save(buf, format=fmt, **params)
    return buf.getvalue(), image.width, image.height

# Example usage
if __name__ == "__main__":
    # Capture screen using the provided function
    import mss
    from PIL import Image

    def capture_screen() -> Image.Image:
        with mss.mss() as sct:
            monitor = sct.monitors[0]  # full screen
            sct_img = sct.grab(monitor)
            img = Image.frombytes("RGB", sct_img.size, sct_img.rgb)
        return img

    screenshot = capture_screen()
    jpeg_bytes, w, h = encode_image(screenshot, fmt="JPEG", quality=85)
    print(f"JPEG size: {len(jpeg_bytes)} bytes, dimensions: {w}x{h}")

    png_bytes, w, h = encode_image(screenshot, fmt="PNG")
    print(f"PNG size: {len(png_bytes)} bytes, dimensions: {w}x{h}")
```

### [3] TCP 伺服器
建立一個簡易 TCP 伺服器，持續接收來自客戶端的連線，並以固定頻率（如 10fps）將編碼後的影像 bytes 傳送。輸入：無。輸出：無。需處理連線斷開與重連。

```python
import socket
import struct
import time
import io
import sys
import mss
from PIL import Image

def capture_screen() -> Image.Image:
    with mss.mss() as sct:
        monitor = sct.monitors[0]
        sct_img = sct.grab(monitor)
        img = Image.frombytes("RGB", sct_img.size, sct_img.rgb)
    return img

def encode_image(image: Image.Image, fmt: str = "JPEG", **params) -> bytes:
    buf = io.BytesIO()
    image.save(buf, format=fmt, **params)
    return buf.getvalue()

def send_image(sock: socket.socket, data: bytes) -> None:
    header = struct.pack(">I", len(data))
    sock.sendall(header + data)

def handle_client(conn: socket.socket, addr):
    print(f"Client connected: {addr}")
    try:
        while True:
            img = capture_screen()
            jpeg_bytes = encode_image(img, fmt="JPEG", quality=85)
            send_image(conn, jpeg_bytes)
            time.sleep(0.1)  # 10 fps
    except (BrokenPipeError, ConnectionResetError, OSError):
        print(f"Client disconnected: {addr}")
    finally:
        conn.close()

def run_server(host: str, port: int):
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        s.bind((host, port))
        s.listen(5)
        print(f"Server listening on {host}:{port}")
        while True:
            try:
                conn, addr = s.accept()
                handle_client(conn, addr)
            except KeyboardInterrupt:
                print("\nServer shutting down.")
                break
            except Exception as e:
                print(f"Error: {e}")

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python server.py <host> <port>")
        sys.exit(1)
    host, port = sys.argv[1], int(sys.argv[2])
    run_server(host, port)
```

### [4] TCP 客戶端
建立一個 TCP 客戶端，連線至伺服器，持續接收影像 bytes，並將其解碼回 PIL Image。輸入：伺服器 IP 與 port。輸出：PIL Image 物件。

```python
import socket
import struct
import io
from PIL import Image

def receive_image(sock: socket.socket) -> Image.Image:
    """
    Receive a single image from the socket.
    The image is prefixed with a 4‑byte big‑endian unsigned integer
    indicating the length of the image data.
    """
    # Read the 4‑byte length header
    header = b""
    while len(header) < 4:
        chunk = sock.recv(4 - len(header))
        if not chunk:
            raise ConnectionError("Connection closed while reading header")
        header += chunk
    (length,) = struct.unpack(">I", header)

    # Read the image data
    data = b""
    while len(data) < length:
        chunk = sock.recv(min(4096, length - len(data)))
        if not chunk:
            raise ConnectionError("Connection closed while reading image data")
        data += chunk

    # Decode the image
    return Image.open(io.BytesIO(data))

def tcp_client(ip: str, port: int):
    """
    Connect to the server and continuously receive images.
    Yields PIL Image objects.
    """
    with socket.create_connection((ip, port)) as sock:
        while True:
            try:
                img = receive_image(sock)
                yield img
            except ConnectionError:
                break

if __name__ == "__main__":
    import sys
    if len(sys.argv) != 3:
        print("Usage: python client.py <server_ip> <port>")
        sys.exit(1)

    server_ip = sys.argv[1]
    server_port = int(sys.argv[2])

    for idx, image in enumerate(tcp_client(server_ip, server_port), 1):
        print(f"Received image {idx}: {image.size}")
        image.show()  # Opens the image in the default viewer
        # To stop after one image, uncomment the next line:
        # break
```

### [5] 畫面顯示
使用 OpenCV 或 Tkinter 在本地視窗中即時顯示從客戶端接收到的 PIL Image。輸入：PIL Image。輸出：視窗顯示。需處理畫面刷新與關閉事件。

```python
import sys
import socket
import struct
import io
import cv2
import numpy as np
from PIL import Image

def receive_image(sock: socket.socket) -> Image.Image:
    header = b""
    while len(header) < 4:
        chunk = sock.recv(4 - len(header))
        if not chunk:
            raise ConnectionError("Connection closed while reading header")
        header += chunk
    (length,) = struct.unpack(">I", header)

    data = b""
    while len(data) < length:
        chunk = sock.recv(min(4096, length - len(data)))
        if not chunk:
            raise ConnectionError("Connection closed while reading image data")
        data += chunk

    return Image.open(io.BytesIO(data))

def tcp_client(ip: str, port: int):
    with socket.create_connection((ip, port)) as sock:
        while True:
            try:
                img = receive_image(sock)
                yield img
            except ConnectionError:
                break

def main():
    if len(sys.argv) != 3:
        print("Usage: python client.py <server_ip> <port>")
        sys.exit(1)

    server_ip = sys.argv[1]
    server_port = int(sys.argv[2])

    cv2.namedWindow("Screen", cv2.WINDOW_NORMAL)
    for idx, pil_img in enumerate(tcp_client(server_ip, server_port), 1):
        cv2_img = cv2.cvtColor(np.array(pil_img), cv2.COLOR_RGB2BGR)
        cv2.imshow("Screen", cv2_img)
        key = cv2.waitKey(1)
        if key == ord('q'):
            break
        if cv2.getWindowProperty("Screen", cv2.WND_PROP_VISIBLE) < 1:
            break
    cv2.destroyAllWindows()

if __name__ == "__main__":
    main()
```

### [6] 簡易驗證
在伺服器與客戶端之間加入一個簡單的 token 驗證機制，確保只有持有正確 token 的客戶端能接收畫面。輸入：token。輸出：驗證結果。

```python
import socket
import struct
import time
import io
import sys
import mss
from PIL import Image
import cv2
import numpy as np

# ---------- Server ----------
def capture_screen() -> Image.Image:
    with mss.mss() as sct:
        monitor = sct.monitors[0]
        sct_img = sct.grab(monitor)
        img = Image.frombytes("RGB", sct_img.size, sct_img.rgb)
    return img

def encode_image(image: Image.Image, fmt: str = "JPEG", **params) -> bytes:
    buf = io.BytesIO()
    image.save(buf, format=fmt, **params)
    return buf.getvalue()

def send_image(sock: socket.socket, data: bytes):
    header = struct.pack(">I", len(data))
    sock.sendall(header + data)

def handle_client(conn: socket.socket, addr, expected_token: str):
    try:
        # Receive token length
        header = b""
        while len(header) < 4:
            chunk = conn.recv(4 - len(header))
            if not chunk:
                return
            header += chunk
        (token_len,) = struct.unpack(">I", header)

        # Receive token
        token_bytes = b""
        while len(token_bytes) < token_len:
            chunk = conn.recv(min(4096, token_len - len(token_bytes)))
            if not chunk:
                return
            token_bytes += chunk
        token = token_bytes.decode()

        if token != expected_token:
            # Send failure message
            msg = b"INVALID_TOKEN"
            send_image(conn, msg)
            return

        # Send success message
        msg = b"OK"
        send_image(conn, msg)

        # Start streaming screenshots
        while True:
            img = capture_screen()
            jpeg_bytes = encode_image(img, fmt="JPEG", quality=85)
            send_image(conn, jpeg_bytes)
            time.sleep(0.1)  # 10 fps
    except Exception:
        pass
    finally:
        conn.close()

def run_server(host: str, port: int, token: str):
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.bind((host, port))
        s.listen(1)
        print(f"Server listening on {host}:{port}")
        while True:
            conn, addr = s.accept()
            print(f"Connection from {addr}")
            handle_client(conn, addr, token)

# ---------- Client ----------
def receive_image(sock: socket.socket) -> bytes:
    header = b""
    while len(header) < 4:
        chunk = sock.recv(4 - len(header))
        if not chunk:
            raise ConnectionError("Connection closed while reading header")
        header += chunk
    (length,) = struct.unpack(">I", header)
    data = b""
    while len(data) < length:
        chunk = sock.recv(min(4096, length - len(data)))
        if not chunk:
            raise ConnectionError("Connection closed while reading data")
        data += chunk
    return data

def run_client(server_ip: str, port: int, token: str):
    with socket.create_connection((server_ip, port)) as sock:
        # Send token
        token_bytes = token.encode()
        sock.sendall(struct.pack(">I", len(token_bytes)) + token_bytes)

        # Receive verification result
        result = receive_image(sock)
        if result == b"INVALID_TOKEN":
            print("Token invalid. Connection closed.")
            return
        elif result == b"OK":
            print("Token verified. Starting to receive images.")
        else:
            print("Unexpected response. Connection closed.")
            return

        cv2.namedWindow("Screen", cv2.WINDOW_NORMAL)
        while True:
            try:
                img_bytes = receive_image(sock)
                img = Image.open(io.BytesIO(img_bytes))
                cv2_img = cv2.cvtColor(np.array(img), cv2.COLOR_RGB2BGR)
                cv2.imshow("Screen", cv2_img)
                if cv2.waitKey(1) & 0xFF == ord('q'):
                    break
            except ConnectionError:
                break
        cv2.destroyAllWindows()

# ---------- Entry Point ----------
if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage:")
        print("  Server: python script.py server <host> <port> <token>")
        print("  Client: python script.py client <server_ip> <port> <token>")
        sys.exit(1)

    mode = sys.argv[1]
    if mode == "server":
        if len(sys.argv) != 5:
            print("Server usage: python script.py server <host> <port> <token>")
            sys.exit(1)
        host, port, token = sys.argv[2], int(sys.argv[3]), sys.argv[4]
        run_server(host, port, token)
    elif mode == "client":
        if len(sys.argv) != 5:
            print("Client usage: python script.py client <server_ip> <port> <token>")
            sys.exit(1)
        server_ip, port, token = sys.argv[2], int(sys.argv[3]), sys.argv[4]
        run_client(server_ip, port, token)
    else:
        print("Unknown mode. Use 'server' or 'client'.")
        sys.exit(1)
```
